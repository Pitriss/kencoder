# Optional Plasma 6 configuration module. The standalone Qt Widgets GUI must
# never link against KF6: these dependencies belong to this plugin alone.
# KF6CoreAddons 6.13 treats an unset BUILD_SHARED_LIBS as a request for a
# STATIC plugin. Such a target builds successfully but is never installed.
# Limit this to the kcm/ subdirectory: the standalone Qt Widgets GUI is
# unaffected and the KCM must be a loadable MODULE library.
set(BUILD_SHARED_LIBS ON)
# KF6CoreAddons appends the plugin namespace to this output directory.
# Without a base directory, the linker incorrectly targets /plasma/kcms/...
# Scope this path to the optional KCM, leaving the Qt Widgets GUI untouched.
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/lib")
kcmutils_add_qml_kcm(kcm_kencoder)
# KDE uses the module filename as its plugin ID when loading the KCM.
# The default CMake 'lib' prefix yields 'libkcm_kencoder', but
# kcmutils_add_qml_kcm embeds the QML at /kcm/kcm_kencoder/main.qml.
# Keep the plugin ID and the embedded resource path consistent.
set_target_properties(kcm_kencoder PROPERTIES PREFIX "")
# Keep the QObject declaration and the plugin factory in the same .cpp.
# The in-source .moc include avoids broken autogenerated header-relative paths
# when the build directory is under the project's .tmp symlink.
set_target_properties(kcm_kencoder PROPERTIES AUTOMOC ON)
target_sources(kcm_kencoder PRIVATE
    kcm_kencoder.cpp
    "${CMAKE_CURRENT_SOURCE_DIR}/../../notify/session_policy.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/../../notify/config.cpp"
)
target_link_libraries(kcm_kencoder PRIVATE
    Qt6::DBus
    Qt6::Quick
    KF6::CoreAddons
    KF6::KCMUtils
    KF6::KCMUtilsQuick
)
